# P5 员工知识库受控 RAG（skb）· 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现员工知识库受控 RAG：向量检索（pgvector + 四层隔离）→ LLM 受限生成（引用强制选自命中 top-k）→ 非 LLM 校验（引用 ID 白名单 + 相似度），并以 `SkbHandler` 实现接入 P3 网关，返回可追溯 `source_refs` 的答案，无依据时明确返回「暂无明确说明」。

**Architecture:** 「检索 → 生成 → 校验」三段（spec §9）。检索用 `EmbeddingProvider` 生成问题向量 + pgvector 余弦检索（强制 tenant filter）；生成用 `LLMProvider`（System Prompt 强约束 + 引用选自命中片段）；校验用**非 LLM** 手段（引用 ID ∈ 命中集合 + 答案/片段向量相似度）。真实 Embedding/LLM（Azure OpenAI）后置，本 plan 用 Mock。

**Tech Stack:** Java 17、Spring（`@Component`/`@Service`）、`JdbcTemplate` + pgvector（原生 SQL）、JUnit + Mockito + Testcontainers（向量检索集成测试）。

**Spec:** `docs/superpowers/specs/2026-08-21-ai-card-software-architecture-design.md`（§9 受控 RAG 边界、§8.4/§8.6 隔离与 pgvector、§9.5 连续问答语义）。

## Global Constraints

- 检索强制带 `customer_id`+`store_id`（spec §8.2/§8.3，四层隔离），**绝不跨租户**。
- 引用必须**选自命中的 top-k 片段 ID**，LLM 不得自造引用（spec §9.3）。
- 校验用非 LLM 手段，整个流程**只有一次 LLM 调用**（spec §9.2）。
- 无依据 / 引用不合法 / 相似度不达标 → 兜底「暂无明确说明」，不编造（spec §7.3）。
- 1.0 连续问答 = 状态连续 + 每轮独立检索，不携带多轮上下文（spec §9.5）。
- embedding 维度 demo 用 1536（联调按真实模型改，见已知边界）。

```java
package com.aicard.skb.store;

import com.aicard.skb.model.Chunk;
import com.aicard.skb.model.Document;
import com.aicard.skb.model.RetrievedChunk;
import com.pgvector.PGvector;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.core.RowMapper;
import org.springframework.jdbc.support.GeneratedKeyHolder;
import org.springframework.jdbc.support.KeyHolder;
import org.springframework.stereotype.Repository;

import java.sql.PreparedStatement;
import java.time.Instant;
import java.util.List;

@Repository
public class KnowledgeStore {

    private final JdbcTemplate jdbc;

    public KnowledgeStore(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public Document insertDocument(Document d) {
        KeyHolder kh = new GeneratedKeyHolder();
        jdbc.update(con -> {
            PreparedStatement ps = con.prepareStatement(
                    "INSERT INTO document(customer_id, store_id, title, version, status, created_at) VALUES (?,?,?,?,?,?)",
                    new String[]{"id"});
            ps.setLong(1, d.customerId());
            ps.setLong(2, d.storeId());
            ps.setString(3, d.title());
            ps.setString(4, d.version());
            ps.setString(5, d.status());
            ps.setObject(6, d.createdAt());
            return ps;
        }, kh);
        long id = kh.getKey().longValue();
        return new Document(id, d.customerId(), d.storeId(), d.title(), d.version(), d.status(), d.createdAt());
    }

    public void insertChunk(Chunk c, List<Float> embedding) {
        PGvector vec = new PGvector(embedding.stream().mapToDouble(Float::doubleValue).toArray());
        jdbc.update(
                "INSERT INTO chunk(document_id, customer_id, store_id, chunk_index, text, embedding, created_at) VALUES (?,?,?,?,?,?,?)",
                c.documentId(), c.customerId(), c.storeId(), c.chunkIndex(), c.text(), vec, Instant.now());
    }

    public List<RetrievedChunk> searchSimilar(Long customerId, Long storeId, List<Float> embedding, int topK) {
        PGvector vec = new PGvector(embedding.stream().mapToDouble(Float::doubleValue).toArray());
        String sql = """
                SELECT c.id, c.text, d.title, d.version,
                       1 - (c.embedding <=> ?::vector) AS similarity
                FROM chunk c JOIN document d ON c.document_id = d.id
                WHERE c.customer_id = ? AND c.store_id = ? AND d.status = 'published'
                ORDER BY c.embedding <=> ?::vector
                LIMIT ?
                """;
        RowMapper<RetrievedChunk> mapper = (rs, i) -> new RetrievedChunk(
                rs.getLong("id"), rs.getString("text"), rs.getString("title"),
                rs.getString("version"), rs.getDouble("similarity"));
        return jdbc.query(sql, mapper, vec, customerId, storeId, vec, topK);
    }
}
```

```java
package com.aicard.skb.service;

import com.aicard.skb.model.*;
import com.aicard.skb.provider.EmbeddingProvider;
import com.aicard.skb.provider.LLMProvider;
import com.aicard.skb.store.KnowledgeStore;
import org.springframework.stereotype.Service;

import java.util.HashSet;
import java.util.List;
import java.util.Set;

@Service
public class SkbService {

    private static final int TOP_K = 5;

    private final EmbeddingProvider embeddings;
    private final KnowledgeStore store;
    private final LLMProvider llm;
    private final double thetaRetr;
    private final double thetaCheck;

    public SkbService(EmbeddingProvider embeddings, KnowledgeStore store, LLMProvider llm,
                      double thetaRetr, double thetaCheck) {
        this.embeddings = embeddings;
        this.store = store;
        this.llm = llm;
        this.thetaRetr = thetaRetr;
        this.thetaCheck = thetaCheck;
    }

    public SkbResult answer(String question, Long customerId, Long storeId) {
        // ① 检索段
        List<Float> qVec = embeddings.embed(question);
        List<RetrievedChunk> hits = store.searchSimilar(customerId, storeId, qVec, TOP_K);
        if (hits.isEmpty() || hits.get(0).similarity() < thetaRetr) {
            return SkbResult.noMatch();
        }

        // ② 生成段（唯一一次 LLM 调用）
        LlmResult llmOut = llm.generate(question, hits);

        // ③ 校验段（非 LLM）
        Set<Long> hitIds = new HashSet<>();
        hits.forEach(c -> hitIds.add(c.chunkId()));
        boolean citedValid = !llmOut.citedChunkIds().isEmpty()
                && hitIds.containsAll(llmOut.citedChunkIds());
        if (!citedValid) {
            return SkbResult.noMatch();
        }

        List<RetrievedChunk> cited = hits.stream()
                .filter(c -> llmOut.citedChunkIds().contains(c.chunkId())).toList();
        double sim = maxSimilarity(llmOut.answer(), cited);
        if (sim < thetaCheck) {
            return SkbResult.noMatch();
        }

        List<String> sourceRefs = cited.stream()
                .map(c -> c.documentTitle() + "@" + c.version() + "#" + c.chunkId()).toList();
        return SkbResult.ok(llmOut.answer(), sourceRefs);
    }

    private double maxSimilarity(String answer, List<RetrievedChunk> cited) {
        if (cited.isEmpty()) {
            return 0.0;
        }
        List<Float> aVec = embeddings.embed(answer);
        double max = 0.0;
        for (RetrievedChunk c : cited) {
            List<Float> cVec = embeddings.embed(c.text());
            max = Math.max(max, cosine(aVec, cVec));
        }
        return max;
    }

    private double cosine(List<Float> a, List<Float> b) {
        double dot = 0, na = 0, nb = 0;
        for (int i = 0; i < a.size(); i++) {
            dot += a.get(i) * b.get(i);
            na += a.get(i) * a.get(i);
            nb += b.get(i) * b.get(i);
        }
        if (na == 0 || nb == 0) {
            return 0.0;
        }
        return dot / (Math.sqrt(na) * Math.sqrt(nb));
    }
}
```

```java
package com.aicard.skb.handler;

import com.aicard.gateway.handler.SkbHandler;
import com.aicard.gateway.protocol.InboundMessage;
import com.aicard.gateway.protocol.OutboundMessage;
import com.aicard.gateway.session.SessionContext;
import com.aicard.provider.api.SpeechProvider;
import com.aicard.provider.api.TranscribeResult;
import com.aicard.provider.api.SynthesizeResult;
import com.aicard.skb.service.SkbService;
import com.aicard.skb.model.SkbResult;
import org.springframework.stereotype.Component;

import java.util.Base64;
import java.util.concurrent.ConcurrentHashMap;

@Component
public class SkbHandlerImpl implements SkbHandler {

    private final SpeechProvider speech;
    private final SkbService service;
    private final ConcurrentHashMap<String, StringBuilder> buffers = new ConcurrentHashMap<>();

    public SkbHandlerImpl(SpeechProvider speech, SkbService service) {
        this.speech = speech;
        this.service = service;
    }

    @Override
    public void onAudioChunk(SessionContext ctx, InboundMessage msg) {
        buffers.computeIfAbsent(msg.turnId(), t -> new StringBuilder())
                .append(msg.audioPayload() == null ? "" : msg.audioPayload());
    }

    @Override
    public void onEndOfUtterance(SessionContext ctx, InboundMessage msg) {
        String turnId = msg.turnId();
        StringBuilder b = buffers.remove(turnId);
        byte[] audio = b == null ? new byte[0] : Base64.getDecoder().decode(b.toString());

        TranscribeResult asr = speech.transcribe(audio, ctx.staffLanguage());
        SkbResult result = service.answer(asr.finalText(), ctx.customerId(), ctx.storeId());

        String answerText = result.answer();
        SynthesizeResult tts = speech.synthesize(answerText, ctx.staffLanguage());
        ctx.send(new OutboundMessage("answer", turnId,
                Base64.getEncoder().encodeToString(tts.ttsAudio()), answerText, null, null, null, null, null));
    }
}
```
