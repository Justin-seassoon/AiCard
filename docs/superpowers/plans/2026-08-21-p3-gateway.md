# P3 设备接入网关（gateway）· 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立设备接入网关：原始 WSS 接入 + 握手鉴权（device_id+token → 推导 tenant）、会话管理（session/mode/turn_id）、**二进制音频帧 + JSON 控制消息**的编解码收发、按模式路由到翻译/问答 handler 接口、设备状态（online/sleep）。

**Architecture:** Spring WebSocket（原始 WSS，非 STOMP）。**音频走 binary frame（`AudioChunkFrame` 上行 / `TtsAudioFrame` 下行，定长帧头 + 大端序），控制消息走 text frame JSON（`InboundMessage`/`OutboundMessage`）**。握手时 `HandshakeInterceptor` 用 `DeviceAuthenticator` 验 device_id+token，从 DB 查出 device 后把 `customer_id/store_id` 写入 `TenantContext`（P1）。`GatewayTransportHandler` 按帧类型分发：binary → 路由音频、text → 路由控制。真实 token 轮换后置。

**Tech Stack:** Java 17、Spring WebSocket、`ByteBuffer`（大端序编解码）、Jackson、JUnit + Mockito。

**Spec:** `docs/superpowers/specs/2026-08-21-ai-card-software-architecture-design.md`（§2 gateway、§4 接口契约、§6.5 turn_id、§8.2 tenant 推导）+ `docs/端云通信接口规范-v1.md`（§2 帧结构、§3/§4 消息）。

## Global Constraints

- 帧结构严格遵循《端云通信接口规范-v1》§2：音频二进制帧 `0x01`（上行）/`0x81`（下行），多字节整数**大端**。
- WSS 握手鉴权在 `HandshakeInterceptor` 完成；会话内消息**不再带 token**。
- tenant 只由 `DeviceAuthenticator` 从设备身份推导写入 `TenantContext`；消息体里的租户字段一律忽略（spec §8.2）。
- `turn_id` 由设备生成，随音频帧/控制消息上行，网关透传不重新分配；`sequence` 仅在 turn 内递增。
- 下行统一：音频走 `TtsAudioFrame`（binary），控制走 `OutboundMessage`（text JSON）。
- P3 不实现翻译/问答业务逻辑，只定义 `TranslationHandler`/`SkbHandler` 接口并提供路由。

package com.aicard.gateway.protocol;

import java.nio.ByteBuffer;
import java.nio.ByteOrder;
import java.nio.charset.StandardCharsets;

public final class AudioFrameCodec {

    public static final byte AUDIO_CHUNK = 0x01;
    public static final byte TTS_AUDIO = (byte) 0x81;

    private AudioFrameCodec() {}

    public static byte[] encodeAudioChunk(AudioChunkFrame f) {
        byte[] turnId = f.turnId().getBytes(StandardCharsets.UTF_8);
        ByteBuffer buf = ByteBuffer.allocate(2 + turnId.length + 4 + 8 + 1 + 1 + f.audio().length)
                .order(ByteOrder.BIG_ENDIAN);
        buf.put(AUDIO_CHUNK);
        buf.put((byte) turnId.length);
        buf.put(turnId);
        buf.putInt(f.sequence());
        buf.putLong(f.timestamp());
        buf.put(f.sourceSide());
        buf.put(f.inputSource());
        buf.put(f.audio());
        return buf.array();
    }

    public static AudioChunkFrame decodeAudioChunk(byte[] frame) {
        ByteBuffer buf = ByteBuffer.wrap(frame).order(ByteOrder.BIG_ENDIAN);
        if (buf.get() != AUDIO_CHUNK) {
            throw new IllegalArgumentException("not an audio chunk frame");
        }
        int turnIdLen = buf.get() & 0xFF;
        byte[] turnId = new byte[turnIdLen];
        buf.get(turnId);
        int sequence = buf.getInt();
        long timestamp = buf.getLong();
        byte sourceSide = buf.get();
        byte inputSource = buf.get();
        byte[] audio = new byte[buf.remaining()];
        buf.get(audio);
        return new AudioChunkFrame(new String(turnId, StandardCharsets.UTF_8),
                sequence, timestamp, sourceSide, inputSource, audio);
    }

    public static byte[] encodeTtsAudio(TtsAudioFrame f) {
        byte[] turnId = f.turnId().getBytes(StandardCharsets.UTF_8);
        ByteBuffer buf = ByteBuffer.allocate(2 + turnId.length + f.audio().length)
                .order(ByteOrder.BIG_ENDIAN);
        buf.put(TTS_AUDIO);
        buf.put((byte) turnId.length);
        buf.put(turnId);
        buf.put(f.audio());
        return buf.array();
    }

    public static TtsAudioFrame decodeTtsAudio(byte[] frame) {
        ByteBuffer buf = ByteBuffer.wrap(frame).order(ByteOrder.BIG_ENDIAN);
        if (buf.get() != TTS_AUDIO) {
            throw new IllegalArgumentException("not a tts audio frame");
        }
        int turnIdLen = buf.get() & 0xFF;
        byte[] turnId = new byte[turnIdLen];
        buf.get(turnId);
        byte[] audio = new byte[buf.remaining()];
        buf.get(audio);
        return new TtsAudioFrame(new String(turnId, StandardCharsets.UTF_8), audio);
    }
}

package com.aicard.gateway.session;

import com.aicard.gateway.protocol.OutboundMessage;

import java.util.function.Consumer;

public class SessionContext {
    private final String sessionId;
    private final String deviceId;
    private final Long customerId;
    private final Long storeId;
    private final String mode;          // translate | qa
    private final String staffLanguage;
    private volatile String turnId;
    private volatile String visitorLanguage;  // P4 维护；UNKNOWN 表示未锁定
    private final Consumer<OutboundMessage> textSender;
    private final Consumer<TtsAudioFrame> binarySender;

    public SessionContext(String sessionId, String deviceId, Long customerId, Long storeId,
                          String mode, String staffLanguage,
                          Consumer<OutboundMessage> textSender, Consumer<TtsAudioFrame> binarySender) {
        this.sessionId = sessionId;
        this.deviceId = deviceId;
        this.customerId = customerId;
        this.storeId = storeId;
        this.mode = mode;
        this.staffLanguage = staffLanguage;
        this.textSender = textSender;
        this.binarySender = binarySender;
        this.visitorLanguage = "UNKNOWN";
    }

    public void sendText(OutboundMessage msg) { textSender.accept(msg); }
    public void sendTtsAudio(TtsAudioFrame frame) { binarySender.accept(frame); }

    public String sessionId() { return sessionId; }
    public String deviceId() { return deviceId; }
    public Long customerId() { return customerId; }
    public Long storeId() { return storeId; }
    public String mode() { return mode; }
    public String staffLanguage() { return staffLanguage; }
    public String turnId() { return turnId; }
    public void turnId(String turnId) { this.turnId = turnId; }
    public String visitorLanguage() { return visitorLanguage; }
    public void visitorLanguage(String v) { this.visitorLanguage = v; }
}

package com.aicard.gateway.session;

import com.aicard.gateway.protocol.OutboundMessage;
import com.aicard.gateway.protocol.TtsAudioFrame;

import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;
import java.util.function.Consumer;

public class SessionManager {

    private final ConcurrentHashMap<String, SessionContext> sessions = new ConcurrentHashMap<>();

    public SessionContext create(String sessionId, String deviceId, Long customerId, Long storeId,
                                 String mode, String staffLanguage,
                                 Consumer<OutboundMessage> textSender, Consumer<TtsAudioFrame> binarySender) {
        SessionContext ctx = new SessionContext(sessionId, deviceId, customerId, storeId,
                mode, staffLanguage, textSender, binarySender);
        sessions.put(sessionId, ctx);
        return ctx;
    }

    public Optional<SessionContext> get(String sessionId) {
        return Optional.ofNullable(sessions.get(sessionId));
    }

    public void remove(String sessionId) {
        sessions.remove(sessionId);
    }
}

package com.aicard.gateway.auth;

import com.aicard.common.domain.Device;
import com.aicard.common.repository.DeviceRepository;

@Component
public class DeviceAuthenticator {

    private final DeviceRepository devices;

    public DeviceAuthenticator(DeviceRepository devices) {
        this.devices = devices;
    }

    public Device authenticate(String deviceId, String token) {
        Device device = devices.findByDeviceId(deviceId)

        }

    }

package com.aicard.gateway.handler;

import com.aicard.gateway.protocol.AudioChunkFrame;
import com.aicard.gateway.protocol.InboundMessage;
import com.aicard.gateway.session.SessionContext;
import com.aicard.gateway.session.SessionManager;
import org.springframework.stereotype.Component;

@Component
public class GatewayWebSocketHandler {

    private final TranslationHandler translation;
    private final SkbHandler skb;
    private final SessionManager sessions;

    public GatewayWebSocketHandler(TranslationHandler translation, SkbHandler skb, SessionManager sessions) {
        this.translation = translation;
        this.skb = skb;
        this.sessions = sessions;
    }

    public void handleAudio(SessionContext ctx, AudioChunkFrame frame) {
        ctx.turnId(frame.turnId());
        if ("qa".equals(ctx.mode())) {
            skb.onAudioChunk(ctx, frame);
        } else {
            translation.onAudioChunk(ctx, frame);
        }
    }

    public void handleControl(SessionContext ctx, InboundMessage msg) {
        if (msg.turnId() != null) {
            ctx.turnId(msg.turnId());
        }
        if ("qa".equals(ctx.mode())) {
            routeControlToSkb(ctx, msg);
        } else {
            routeControlToTranslation(ctx, msg);
        }
    }

    private void routeControlToTranslation(SessionContext ctx, InboundMessage msg) {
        switch (msg.type()) {
            case "eou" -> translation.onEndOfUtterance(ctx, msg);
            case "stop_tts" -> translation.onStopTts(ctx, msg);
            default -> { /* sleep_notice / button_event / playback_event 由 Task 5 处理 */ }
        }
    }

    private void routeControlToSkb(SessionContext ctx, InboundMessage msg) {
        if ("eou".equals(msg.type())) {
            skb.onEndOfUtterance(ctx, msg);
        }
    }
}

package com.aicard.gateway.handler;

import com.aicard.common.domain.Device;
import com.aicard.common.tenant.Tenant;
import com.aicard.common.tenant.TenantContext;
import com.aicard.gateway.auth.DeviceAuthenticator;
import com.aicard.gateway.protocol.*;
import com.aicard.gateway.session.SessionContext;
import com.aicard.gateway.session.SessionManager;
import com.aicard.gateway.state.DeviceStateTracker;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.web.socket.BinaryMessage;
import org.springframework.web.socket.TextMessage;
import org.springframework.web.socket.WebSocketSession;
import org.springframework.web.socket.handler.AbstractWebSocketHandler;

import java.nio.ByteBuffer;
import java.util.Map;

public class GatewayTransportHandler extends AbstractWebSocketHandler {

    private final ObjectMapper mapper = new ObjectMapper();
    private final DeviceAuthenticator authenticator;
    private final SessionManager sessions;
    private final GatewayWebSocketHandler router;
    private final DeviceStateTracker states;

    public GatewayTransportHandler(DeviceAuthenticator authenticator, SessionManager sessions,
                                   GatewayWebSocketHandler router, DeviceStateTracker states) {
        this.authenticator = authenticator;
        this.sessions = sessions;
        this.router = router;
        this.states = states;
    }

    @Override
    protected void handleBinaryMessage(WebSocketSession session, BinaryMessage message) throws Exception {
        ByteBuffer payload = message.getPayload();
        byte[] bytes = new byte[payload.remaining()];
        payload.get(bytes);
        AudioChunkFrame frame = AudioFrameCodec.decodeAudioChunk(bytes);

        SessionContext ctx = currentContext(session, frame.turnId());
        withTenant(ctx, () -> {
            states.markOnline(ctx.deviceId());
            router.handleAudio(ctx, frame);
        });
    }

    @Override
    protected void handleTextMessage(WebSocketSession session, TextMessage message) throws Exception {
        InboundMessage msg = mapper.readValue(message.getPayload(), InboundMessage.class);
        if ("session_init".equals(msg.type())) {
            initSession(session, msg);
            return;
        }
        SessionContext ctx = currentContext(session, msg.sessionId());
        withTenant(ctx, () -> {
            if ("sleep_notice".equals(msg.type())) {
                states.markSleep(ctx.deviceId());
                return;
            }
            states.markOnline(ctx.deviceId());
            router.handleControl(ctx, msg);
        });
    }

    private void initSession(WebSocketSession session, InboundMessage msg) {
        Device device = (Device) session.getAttributes().get("device");
        sessions.create(msg.sessionId(), device.getDeviceId(), device.getCustomerId(),
                device.getStoreId(),
                msg.mode() == null ? "translate" : msg.mode(),
                msg.staffLanguage() == null ? "ja-JP" : msg.staffLanguage(),
                out -> sendText(session, out),
                frame -> sendBinary(session, frame));
        states.markOnline(device.getDeviceId());
    }

    private SessionContext currentContext(WebSocketSession session, String sessionId) {
        Map<String, Object> attrs = session.getAttributes();
        Device device = (Device) attrs.get("device");
        String mode = (String) attrs.getOrDefault("mode", "translate");
        String staffLanguage = (String) attrs.getOrDefault("staff_language", "ja-JP");
        return sessions.get(sessionId).orElseGet(() ->
                sessions.create(sessionId, device.getDeviceId(), device.getCustomerId(),
                        device.getStoreId(), mode, staffLanguage,
                        out -> sendText(session, out),
                        frame -> sendBinary(session, frame)));
    }

    private void withTenant(SessionContext ctx, Runnable action) {
        TenantContext.set(Tenant.of(ctx.customerId(), ctx.storeId()));
        try {
            action.run();
        } finally {
            TenantContext.clear();
        }
    }

    private void sendText(WebSocketSession session, OutboundMessage out) {
        try {
            session.sendMessage(new TextMessage(mapper.writeValueAsString(out)));
        } catch (Exception e) {
            throw new RuntimeException("downstream text send failed", e);
        }
    }

    private void sendBinary(WebSocketSession session, TtsAudioFrame frame) {
        try {
            session.sendMessage(new BinaryMessage(AudioFrameCodec.encodeTtsAudio(frame)));
        } catch (Exception e) {
            throw new RuntimeException("downstream binary send failed", e);
        }
    }
}

package com.aicard.gateway.auth;

import com.aicard.common.domain.Device;
import org.springframework.http.server.ServerHttpRequest;
import org.springframework.http.server.ServerHttpResponse;
import org.springframework.http.server.ServletServerHttpRequest;
import org.springframework.web.socket.WebSocketHandler;
import org.springframework.web.socket.server.HandshakeInterceptor;

import java.util.Map;

public class GatewayHandshakeInterceptor implements HandshakeInterceptor {

    private final DeviceAuthenticator authenticator;

    public GatewayHandshakeInterceptor(DeviceAuthenticator authenticator) {
        this.authenticator = authenticator;
    }

    @Override
    public boolean beforeHandshake(ServerHttpRequest request, ServerHttpResponse response,
                                   WebSocketHandler wsHandler, Map<String, Object> attributes) {
        if (!(request instanceof ServletServerHttpRequest servletRequest)) {
            return false;
        }
        String deviceId = servletRequest.getServletRequest().getParameter("device_id");
        String token = servletRequest.getServletRequest().getParameter("token");
        String fwVersion = servletRequest.getServletRequest().getParameter("fw_version");
        if (deviceId == null || token == null) {
            return false;
        }
        Device device = authenticator.authenticate(deviceId, token);
        attributes.put("device", device);
        attributes.put("fw_version", fwVersion);
        return true;
    }

    @Override
    public void afterHandshake(ServerHttpRequest request, ServerHttpResponse response,
                               WebSocketHandler wsHandler, Exception exception) {
    }
}
